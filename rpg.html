const boostConfig = { baseSpeed: 5, boostMultiplier: 1.8, baseDuration: 2.0, // seconds baseCooldown: 15.0, // seconds }; let player = { speed: boostConfig.baseSpeed, isBoosting: false, canBoost: true, boostTimer: 0, cooldownTimer: 0 }; function triggerBoost(skillLevel = 0) { if (!player.canBoost) return; // Calculate stats based on skill level const activeDuration = boostConfig.baseDuration + (skillLevel * 0.5); // +0.5s per rank const activeCooldown = Math.max(5, boostConfig.baseCooldown - (skillLevel * 2.0)); // -2s per rank player.isBoosting = true; player.canBoost = false; player.speed = boostConfig.baseSpeed * boostConfig.boostMultiplier; // End boost after active duration setTimeout(() => { player.isBoosting = false; player.speed = boostConfig.baseSpeed; }, activeDuration * 1000); // Cooldown countdown player.cooldownTimer = activeCooldown; const cdInterval = setInterval(() => { player.cooldownTimer -= 0.1; updateBoostUI(player.cooldownTimer, activeCooldown); if (player.cooldownTimer <= 0) { clearInterval(cdInterval); player.canBoost = true; player.cooldownTimer = 0; updateBoostUI(0, activeCooldown); } }, 100); }
